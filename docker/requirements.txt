# Python dependencies for the CUDA transcription container.
#
# torch / torchaudio are NOT listed here: they ship in the pytorch base image
# and are pinned at build time via a generated constraints file so no
# transitive dependency can silently swap in a different (CPU-only) build.

# --- ASR / NLP ---
transformers>=4.57,<5
accelerate>=1.0
peft>=0.14                # IBM Granite Speech ships its ASR head as a LoRA adapter

# --- Diarization ---
# speaker-diarization-community-1 requires the pyannote.audio 4.x runtime.
pyannote.audio>=4.0,<5

# pyannote 4.x decodes audio through torchcodec. torchcodec declares NO torch
# dependency and its PyPI wheels are built against CUDA 13, so the base image
# must be a CUDA 13 one or its .so fails to dlopen libnvrtc.so.13.
torchcodec>=0.16

# --- Audio I/O ---
librosa>=0.10
soundfile>=0.12
numpy>=1.26

# --- Misc ---
